cmake_minimum_required(VERSION 3.20)
project(PointCloudWorkbench VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
if(MSVC)
  add_compile_options(/utf-8 /bigobj /wd4251 /wd4275)
  add_compile_definitions(NOMINMAX _USE_MATH_DEFINES)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/$<CONFIG>")
find_package(Qt5 5.12 REQUIRED COMPONENTS Core Gui Widgets OpenGL Test)
find_package(PCL 1.11 REQUIRED COMPONENTS common io filters segmentation visualization)
if(NOT PCL_VERSION VERSION_LESS 1.12)
  message(FATAL_ERROR "This SDK requires PCL 1.11.x")
endif()
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc)
find_package(VTK REQUIRED)
if(VTK_VERSION VERSION_LESS 9)
  include(${VTK_USE_FILE})
  set(FW_VTK_QT vtkGUISupportQt)
  set(FW_VTK_RENDER vtkRenderingOpenGL2 vtkInteractionStyle vtkRenderingFreeType)
else()
  find_package(VTK REQUIRED COMPONENTS GUISupportQt RenderingOpenGL2 InteractionStyle)
  set(FW_VTK_QT VTK::GUISupportQt)
  set(FW_VTK_RENDER VTK::RenderingOpenGL2 VTK::InteractionStyle VTK::RenderingFreeType)
endif()
if(NOT TARGET ${FW_VTK_QT})
  message(FATAL_ERROR "The VTK used by PCL must include Qt support")
endif()
include(CMakePackageConfigHelpers)
include(GNUInstallDirs)
set(FW_ABI_BASE "sdk1|${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}|${CMAKE_SIZEOF_VOID_P}|qt${Qt5Core_VERSION}|pcl${PCL_VERSION}|vtk${VTK_VERSION}|cv${OpenCV_VERSION}|cxx17|sharedCRT")
configure_file(cmake/BuildInfo.h.in generated/fw/BuildInfo.h @ONLY)
file(WRITE "${CMAKE_BINARY_DIR}/dependencies.lock.txt" "${FW_ABI_BASE}\nPCL_DIR=${PCL_DIR}\nVTK_DIR=${VTK_DIR}\nQt5_DIR=${Qt5_DIR}\nOpenCV_DIR=${OpenCV_DIR}\n")
include(cmake/FrameworkPlugin.cmake)

add_library(FrameworkSDK SHARED sdk/src/Parameters.cpp sdk/src/Data.cpp sdk/src/Diagnostics.cpp sdk/include/fw/Host.h)
add_library(Framework::SDK ALIAS FrameworkSDK)
set_target_properties(FrameworkSDK PROPERTIES EXPORT_NAME SDK)
target_compile_definitions(FrameworkSDK PRIVATE FW_SDK_BUILD PUBLIC ${PCL_DEFINITIONS})
target_include_directories(FrameworkSDK PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/sdk/include> $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/generated> $<INSTALL_INTERFACE:include>)
foreach(dependency_include IN LISTS PCL_INCLUDE_DIRS OpenCV_INCLUDE_DIRS)
  target_include_directories(FrameworkSDK PUBLIC "$<BUILD_INTERFACE:${dependency_include}>")
endforeach()
target_link_libraries(FrameworkSDK PUBLIC Qt5::Core pcl_common ${OpenCV_LIBS})

add_library(FrameworkCore SHARED core/Runtime.cpp core/Runtime.h core/TrialSession.cpp core/TrialSession.h core/DebugController.cpp core/DebugController.h)
target_compile_definitions(FrameworkCore PRIVATE FW_CORE_BUILD)
target_include_directories(FrameworkCore PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/core>)
target_link_libraries(FrameworkCore PUBLIC FrameworkSDK PRIVATE pcl_io)

framework_add_plugin(ParameterUI TYPE ui ID framework.parameters SOURCES plugins/parameters/ParameterUI.cpp)
target_link_libraries(ParameterUI PRIVATE Qt5::Widgets)
framework_add_plugin(Visualization TYPE ui ID framework.visualization SOURCES plugins/visualization/Visualization.cpp)
target_link_libraries(Visualization PRIVATE Qt5::Widgets Qt5::OpenGL pcl_visualization ${FW_VTK_QT} ${FW_VTK_RENDER})
framework_add_plugin(MainUI TYPE ui ID framework.main SOURCES plugins/main/MainUI.cpp)
target_link_libraries(MainUI PRIVATE FrameworkCore Qt5::Widgets)
add_subdirectory(algorithms/sample)
add_subdirectory(algorithms/corrugated_panel)

add_executable(Launcher WIN32 app/main.cpp ui/icons.qrc)
foreach(ui_target Launcher MainUI ParameterUI Visualization)
  target_include_directories(${ui_target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/ui")
endforeach()
target_link_libraries(Launcher PRIVATE FrameworkCore Qt5::Widgets ${FW_VTK_QT} ${FW_VTK_RENDER})
add_dependencies(Launcher MainUI ParameterUI Visualization SampleAlgorithm CorrugatedPanelAlgorithm)
if(COMMAND vtk_module_autoinit)
  vtk_module_autoinit(TARGETS Visualization Launcher MODULES ${VTK_LIBRARIES})
endif()

install(TARGETS FrameworkSDK EXPORT FrameworkSDKTargets RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(TARGETS FrameworkCore Launcher RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY sdk/include/ DESTINATION include)
install(FILES "${CMAKE_BINARY_DIR}/generated/fw/BuildInfo.h" DESTINATION include/fw)
install(EXPORT FrameworkSDKTargets NAMESPACE Framework:: DESTINATION lib/cmake/FrameworkSDK)
configure_package_config_file(cmake/FrameworkSDKConfig.cmake.in "${CMAKE_BINARY_DIR}/FrameworkSDKConfig.cmake" INSTALL_DESTINATION lib/cmake/FrameworkSDK)
write_basic_package_version_file("${CMAKE_BINARY_DIR}/FrameworkSDKConfigVersion.cmake" VERSION 1.0.0 COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_BINARY_DIR}/FrameworkSDKConfig.cmake" "${CMAKE_BINARY_DIR}/FrameworkSDKConfigVersion.cmake" cmake/FrameworkPlugin.cmake DESTINATION lib/cmake/FrameworkSDK)
install(FILES "${CMAKE_BINARY_DIR}/dependencies.lock.txt" DESTINATION .)
install(DIRECTORY examples/ DESTINATION examples)
install(DIRECTORY docs/ DESTINATION docs)
install(DIRECTORY schemas/ DESTINATION schemas)
install(FILES AGENTS.md DESTINATION .)
include(CTest)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
if(WIN32)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS_SKIP TRUE)
  include(InstallRequiredSystemLibraries)
  get_target_property(FW_QMAKE Qt5::qmake IMPORTED_LOCATION)
  get_filename_component(FW_QT_BIN "${FW_QMAKE}" DIRECTORY)
  get_filename_component(FW_PCL_ROOT "${PCL_DIR}/.." ABSOLUTE)
  get_filename_component(FW_OPENCV_BIN "${OpenCV_DIR}/../bin" ABSOLUTE)
  get_filename_component(FW_COMPILER_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
  set(FW_RUNTIME_DIRS "" CACHE STRING "Additional runtime DLL search directories")
  configure_file(cmake/DeployRuntime.cmake.in "${CMAKE_BINARY_DIR}/DeployRuntime.cmake" @ONLY)
  add_custom_target(deploy COMMAND ${CMAKE_COMMAND} "-DFW_BIN=$<TARGET_FILE_DIR:Launcher>" "-DFW_CONFIG=$<CONFIG>" -P "${CMAKE_BINARY_DIR}/DeployRuntime.cmake" DEPENDS Launcher)
endif()
