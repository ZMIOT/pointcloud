cmake_minimum_required(VERSION 3.20)
project(ExtractFeature2Plugin VERSION 1.9.0 LANGUAGES C CXX)
find_package(FrameworkSDK 1 CONFIG REQUIRED)
find_package(PCL 1.11 REQUIRED COMPONENTS common filters io kdtree)
find_path(EXTRACT2_FLANN_INCLUDE flann/util/params.h
  HINTS "${PCL_DIR}/../3rdParty/FLANN/include" ${FLANN_INCLUDE_DIRS} REQUIRED)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options(/utf-8 /bigobj)
  add_compile_definitions(NOMINMAX _USE_MATH_DEFINES)
endif()

# Optional integration harness. It loads existing host DLLs without rebuilding them.
option(EXTRACT_FEATURE2_GUI_SMOKE "Build the host integration smoke tool" OFF)
if(EXTRACT_FEATURE2_GUI_SMOKE)
  find_package(Qt5 REQUIRED COMPONENTS Widgets OpenGL)
  find_package(VTK REQUIRED)
  if(VTK_VERSION VERSION_LESS 9)
    include(${VTK_USE_FILE})
    set(EXTRACT2_VTK_QT vtkGUISupportQt)
  else()
    set(EXTRACT2_VTK_QT VTK::GUISupportQt)
  endif()
  get_filename_component(EXTRACT2_SDK_ROOT "${FrameworkSDK_DIR}/../../.." ABSOLUTE)
  set(EXTRACT_FEATURE2_HOST_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Host headers used only by the integration harness")
  add_executable(ExtractFeature2UiSmoke tests/UiSmoke.cpp "${EXTRACT_FEATURE2_HOST_SOURCE}/ui/icons.qrc")
  set_target_properties(ExtractFeature2UiSmoke PROPERTIES AUTORCC ON)
  target_include_directories(ExtractFeature2UiSmoke PRIVATE "${EXTRACT_FEATURE2_HOST_SOURCE}/core" "${EXTRACT_FEATURE2_HOST_SOURCE}/ui")
  target_link_libraries(ExtractFeature2UiSmoke PRIVATE ExtractFeature2Core Framework::SDK Qt5::Widgets Qt5::OpenGL "${EXTRACT2_SDK_ROOT}/lib/FrameworkCore.lib" ${EXTRACT2_VTK_QT} ${VTK_LIBRARIES})
endif()

add_library(ExtractFeature2Core STATIC
  core/SeamDetector.cpp core/SeamSmoother.cpp core/SeamHistory.cpp core/PointCloudIO.cpp core/HullSurfaces.cpp)
target_include_directories(ExtractFeature2Core PUBLIC core ${PCL_INCLUDE_DIRS} ${EXTRACT2_FLANN_INCLUDE})
target_link_libraries(ExtractFeature2Core PUBLIC Qt5::Core pcl_common pcl_filters pcl_io pcl_kdtree)

framework_add_plugin(ExtractFeature2Algorithm TYPE algorithm
  ID weld.extract_feature2 ALGORITHM_ID weld.extract_feature2
  VERSION 1.9.0 PARAMETER_VERSION 8 SOURCES Plugin.cpp parameters.qrc)
target_link_libraries(ExtractFeature2Algorithm PRIVATE ExtractFeature2Core)

include(CTest)
if(BUILD_TESTING)
  add_executable(ExtractFeature2PluginTests tests/PluginTests.cpp)
  target_link_libraries(ExtractFeature2PluginTests PRIVATE ExtractFeature2Core Framework::SDK)
  target_compile_definitions(ExtractFeature2PluginTests PRIVATE PLUGIN_PATH="$<TARGET_FILE:ExtractFeature2Algorithm>")
  add_dependencies(ExtractFeature2PluginTests ExtractFeature2Algorithm)
  add_test(NAME plugin_extract2_integration COMMAND ExtractFeature2PluginTests)
  add_executable(ExtractFeature2SurfaceTests tests/SurfaceTests.cpp)
  target_link_libraries(ExtractFeature2SurfaceTests PRIVATE ExtractFeature2Core Framework::SDK)
  target_compile_definitions(ExtractFeature2SurfaceTests PRIVATE PLUGIN_PATH="$<TARGET_FILE:ExtractFeature2Algorithm>")
  add_dependencies(ExtractFeature2SurfaceTests ExtractFeature2Algorithm)
  add_test(NAME plugin_extract2_surfaces COMMAND ExtractFeature2SurfaceTests)
  add_executable(ExtractFeature2FullSeamTests tests/FullSeamTests.cpp)
  target_link_libraries(ExtractFeature2FullSeamTests PRIVATE ExtractFeature2Core Framework::SDK)
  target_compile_definitions(ExtractFeature2FullSeamTests PRIVATE PLUGIN_PATH="$<TARGET_FILE:ExtractFeature2Algorithm>")
  add_dependencies(ExtractFeature2FullSeamTests ExtractFeature2Algorithm)
  add_test(NAME plugin_extract2_full_seams COMMAND ExtractFeature2FullSeamTests)
  get_filename_component(EXTRACT2_SDK_ROOT "${FrameworkSDK_DIR}/../../.." ABSOLUTE)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../core/Runtime.h" AND EXISTS "${EXTRACT2_SDK_ROOT}/lib/FrameworkCore.lib")
    add_executable(ExtractFeature2TrialSmoke tests/TrialSmoke.cpp)
    target_include_directories(ExtractFeature2TrialSmoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../core")
    target_link_libraries(ExtractFeature2TrialSmoke PRIVATE Framework::SDK "${EXTRACT2_SDK_ROOT}/lib/FrameworkCore.lib")
  endif()
endif()

  add_executable(ExtractFeature2Convert tools/Convert.cpp)
  target_link_libraries(ExtractFeature2Convert PRIVATE ExtractFeature2Core)
